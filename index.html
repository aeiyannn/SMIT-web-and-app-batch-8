        updatebtn.addEventListener("click", () => {
            isUpdated = true;
            Inputvalue.value = text.textContent;
            ele.textContent = "Update";
            ele.addEventListener("click", () => {
                text.textContent = Inputvalue.value;
                ele.textContent = "Add +";
                isUpdated = false;
            });
        });

        delbtn.textContent = "Delete";
        updatebtn.innerText = "Update";
        li.append(text);
        li.append(delbtn);
        li.append(updatebtn);
        todoList.append(li);
        Inputvalue.value = "";
    }
}
